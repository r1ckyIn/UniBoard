    **Step 7: Create app/[locale]/layout.tsx**
    ```typescript
    import { NextIntlClientProvider } from "next-intl";
    import { getMessages, setRequestLocale } from "next-intl/server";
    import { notFound } from "next/navigation";
    import { routing } from "@/lib/i18n/routing";

    type Props = {
      children: React.ReactNode;
      params: Promise<{ locale: string }>;
    };

    export function generateStaticParams() {
      return routing.locales.map((locale) => ({ locale }));
    }

    export default async function LocaleLayout({ children, params }: Props) {
      const { locale } = await params;

      if (!routing.locales.includes(locale as "en" | "zh")) {
        notFound();
      }

      setRequestLocale(locale);
      const messages = await getMessages();

      return (
        <NextIntlClientProvider messages={messages}>
          {children}
        </NextIntlClientProvider>
      );
    }
    ```

    Create a test that verifies en.json and zh.json have identical key structures (every key in EN exists in ZH and vice versa). This prevents translation drift as page phases add new keys.

    ```typescript
    import { describe, it, expect } from "vitest";
    import en from "../../messages/en.json";
    import zh from "../../messages/zh.json";

    function getKeys(obj: Record<string, unknown>, prefix = ""): string[] {
      return Object.entries(obj).flatMap(([key, value]) => {
        const fullKey = prefix ? `${prefix}.${key}` : key;
        if (typeof value === "object" && value !== null) {
          return getKeys(value as Record<string, unknown>, fullKey);
        }
        return [fullKey];
      });
    }

    describe("i18n message key parity", () => {
      it("en.json and zh.json have identical key sets", () => {
        const enKeys = getKeys(en).sort();
        const zhKeys = getKeys(zh).sort();
        expect(enKeys).toEqual(zhKeys);
      });

      it("no empty string values in en.json", () => {
        const enKeys = getKeys(en);
        enKeys.forEach((key) => {
          const value = key.split(".").reduce((obj: Record<string, unknown>, k) => obj[k] as Record<string, unknown>, en as Record<string, unknown>);
          expect(value, `en.json key "${key}" is empty`).not.toBe("");
        });
      });

      it("no empty string values in zh.json", () => {
        const zhKeys = getKeys(zh);
        zhKeys.forEach((key) => {
          const value = key.split(".").reduce((obj: Record<string, unknown>, k) => obj[k] as Record<string, unknown>, zh as Record<string, unknown>);
          expect(value, `zh.json key "${key}" is empty`).not.toBe("");
        });
      });
    });
    ```
