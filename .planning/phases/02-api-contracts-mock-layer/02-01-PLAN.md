  children: React.ReactNode;
  params: Promise<{ locale: string }>;
};

export default async function LocaleLayout({ children, params }: Props) {
  const { locale } = await params;
  if (!hasLocale(routing.locales, locale)) { notFound(); }
  setRequestLocale(locale);
  const messages = await getMessages();

   - `POST /auth/register` — body: {email, password, display_name}, response: {user_id, pending_verification}
   - `POST /auth/login` — body: {email, password}, response: {access_token, refresh_token, expires_in, user}

   - `POST /sync/trigger` — body: {scope}, response: 202 SyncStatus

   - `GpaPrediction`: {current_wam, predicted_wam, delta, per_course: GpaPredictionCourse[]}
   - `GpaPredictionCourse`: {course_id, code, current_mark, predicted_mark, applied_assumptions: WhatIfScore[]}
   - `WhatIfScore`: {course_id, assessment_name, assumed_score}
   - `GpaPath`: {target_wam, current_wam, is_achievable, per_course: GpaPathCourse[]}

   - `GpaPathAssessment`: {name, weight, minimum_score}
   - `DigestLatest`: {digest_id, generated_at, period, courses: DigestCourseEntry[], urgent_deadlines: UrgentDeadline[]}
   - `DigestCourseEntry`: {code, highlights: DigestHighlight[]}
   - `DigestHighlight`: {type, summary, urgency, source_thread_id?}
   - `UrgentDeadline`: {course_code, title, due_date, hours_remaining}
   - `DigestSummary`: {digest_id, generated_at, period, highlight_count}
   - `Alert`: {id, type, severity, course_code, message, current_mark?, target_mark?, deadline?, created_at, is_read}
   - `Notification`: {id, type, severity, title, body, is_read, action_url?, created_at}
   - `SyncTriggerResponse`: {sync_id, status, started_at}
   - `SyncStatusResponse`: {last_sync: SyncDetail, platforms: PlatformStatus}
   - `SyncDetail`: {sync_id, status, started_at, completed_at?, results: SyncResults}
   - `SyncResults`: {grades?: SyncCount, deadlines?: SyncCount, discussions?: SyncCount}
   - `SyncCount`: {synced, new, updated}
   - `PlatformStatus`: {canvas: PlatformHealth, ed: PlatformHealth}
   - `PlatformHealth`: {status: "healthy"|"degraded"|"error", last_success}
