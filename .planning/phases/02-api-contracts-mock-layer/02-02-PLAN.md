```typescript
import { mockDelay, mockResponse, mockError, requireAuth, shouldSimulateError } from "@/lib/fixtures/helpers";

  await mockDelay();
  if (shouldSimulateError()) {
    return mockError("INTERNAL_ERROR", "Mock server error", 500);
  }
  const authError = requireAuth(request);
  if (authError) return authError;
  // ... return mockResponse(data)
}
```

**Auth endpoints (NO auth required):**

1. `POST /auth/register` — Parse body {email, password, display_name}. Validate email contains "@". Return `mockResponse({ user_id: "usr_001", pending_verification: true })` with status 201. On validation fail, return `mockError("VALIDATION_ERROR", "Invalid email format", 400)`.

2. `POST /auth/login` — Parse body {email, password}. Accept any email/password combo for mock. Return `mockResponse({ access_token: "mock-jwt-...", refresh_token: "mock-jwt-refresh-...", expires_in: 900, user: mockUser })`.

3. `POST /auth/refresh` — Parse body {refresh_token}. Return new mock tokens.

4. `POST /auth/logout` — Return `NextResponse.json(null, { status: 204 })`.

5. `POST /auth/forgot-password` — Return `mockResponse({ message: "Password reset code sent" })`.

6. `POST /auth/confirm-password` — Return `mockResponse({ message: "Password reset successful" })`.

Note: forgot-password and confirm-password share the same route file at `frontend/app/api/v1/auth/forgot-password/route.ts` and `frontend/app/api/v1/auth/confirm-password/route.ts` respectively. However, since the files_modified list only includes the 4 main auth files, include forgot-password and confirm-password logic as part of the register route or create separate files as needed. Actually, create them as 2 additional files — they are small enough to not impact the file count significantly and keep the route structure clean. (Total: 6 auth route files)

**User endpoints (auth required):**

**For Route Handlers with dynamic segments** like `[platform]`, use the Next.js App Router convention:
```typescript
export async function PUT(
  request: Request,
  { params }: { params: Promise<{ platform: string }> }
) {
  const { platform } = await params;
  // ...
}
```
Note: In Next.js 15, `params` is a Promise and must be awaited.

**File organization:** Route Handlers go at `frontend/app/api/v1/...` (NOT inside `[locale]`). The existing middleware matcher `["/((?!api|_next|.*\\..*).*)"]` already excludes `/api/` from i18n processing.
