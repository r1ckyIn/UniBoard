```typescript
import { mockDelay, mockResponse, mockError, requireAuth, shouldSimulateError } from "@/lib/fixtures/helpers";
import { fixtureData } from "@/lib/fixtures/domain";

export async function GET(request: Request) {
  await mockDelay();
  if (shouldSimulateError()) {
    return mockError("INTERNAL_ERROR", "Mock server error", 500);
  }
  const authError = requireAuth(request);
  if (authError) return authError;
  // ... return mockResponse(data)
}
```

**ALL endpoints below require auth (call requireAuth first).**

**Course endpoints:**

1. `GET /courses` — Read `semester` query param from `new URL(request.url).searchParams`. Filter courses fixture by semester if provided. Return `mockResponse(filtered)`.

2. `GET /courses/{id}` — Look up courseDetails by id param. Return `mockResponse(detail)` or `mockError("NOT_FOUND", "Course not found", 404)` if ID not in fixture map.

3. `GET /courses/{id}/grades` — Look up gradesByCourse by id. Return `mockResponse(grades)` or empty array if no grades.

4. `GET /courses/{id}/materials` — Look up materialsByCourse. Filter by `source` query param if present ("canvas" or "ed"). Return `mockResponse(materials)`.

5. `GET /courses/{id}/discussions` — Look up discussionsByCourse. Support `filter` query param:
   - "high_value": gpa_relevance_score >= 0.7
   - "endorsed": is_endorsed === true
   - "staff": is_staff_post === true
   - "all" or missing: return all
   Apply cursor/limit pagination via mockPaginatedResponse.

6. `GET /courses/{id}/deadlines` — Filter deadlines fixture by course_code matching the course's code (look up course by id first). Return `mockResponse(filtered)`.

7. `GET /courses/{id}/outline` — Return a mock outline object for the course: `{ sections: [{ title: "Learning Outcomes", content: "..." }, ...], assessment_table: courseDetails[id].assessment_weights, last_updated: "2026-02-15" }`.

**GPA endpoints:**

8. `GET /gpa` — Return `mockResponse(gpaReport)`.

9. `POST /gpa/predict` — Parse body {what_if_scores, scale}. Return `mockResponse(gpaPredictionDefault)`. (In a real implementation, the prediction would compute from what_if_scores; in mock, return the default fixture.)

10. `POST /gpa/path` — Parse body {target_wam}. Return `mockResponse(gpaPathDefault)`.

**Deadline endpoints:**

11. `GET /deadlines` — Support `from`, `to`, `course_id` query params for filtering deadlines array. Parse date strings for from/to comparison against due_date. Return `mockResponse(filtered)`.

12. `GET /deadlines/upcoming` — Filter deadlines where days_remaining is between 0 and 7 (inclusive). Return `mockResponse(filtered)`.

**For Route Handlers with dynamic segments** like `[id]`, use the Next.js 15 App Router convention:
```typescript
export async function GET(
  request: Request,
  { params }: { params: Promise<{ id: string }> }
) {
  const { id } = await params;
  // ...
}
```
Note: In Next.js 15, `params` is a Promise and must be awaited.
