       ```typescript
       import { z } from "zod/v4";

       export const loginSchema = z.object({
         email: z
           .string()
           .min(1, "Email is required")
           .email("Invalid email format")
           .refine((e) => e.endsWith("@uni.sydney.edu.au"), {
             message: "Please use your USYD student email (@uni.sydney.edu.au)",
           }),
         password: z.string().min(1, "Password is required"),
       });

       export const registerSchema = z
         .object({
           displayName: z.string().min(1, "Display name is required"),
           email: z
             .string()
             .min(1, "Email is required")
             .email("Invalid email format")
             .refine((e) => e.endsWith("@uni.sydney.edu.au"), {
               message: "Please use your USYD student email (@uni.sydney.edu.au)",
             }),
           password: z.string().min(8, "Password must be at least 8 characters"),
           confirmPassword: z.string().min(1, "Please confirm your password"),
         })
         .refine((data) => data.password === data.confirmPassword, {
           message: "Passwords do not match",
           path: ["confirmPassword"],
         });

       export type LoginInput = z.infer<typeof loginSchema>;
       export type RegisterInput = z.infer<typeof registerSchema>;

       export function getPasswordStrength(pw: string): 0 | 1 | 2 | 3 | 4 {
         let score = 0;
         if (pw.length >= 8) score++;
         if (pw.length >= 12) score++;
         if (/[A-Z]/.test(pw) && /[a-z]/.test(pw)) score++;
         if (/[0-9]/.test(pw)) score++;
         if (/[^A-Za-z0-9]/.test(pw)) score++;
         return Math.min(score, 4) as 0 | 1 | 2 | 3 | 4;
       }
       ```

       NOTE: zod v4 uses `import { z } from "zod/v4"` — check the installed version. If pnpm installs zod 4.x, use `"zod/v4"`. If zod 3.x is installed, use `"zod"`. Verify with `cat node_modules/zod/package.json | grep version`.

    3. Create `frontend/components/auth/AuthGuard.tsx`:
       ```typescript
       "use client";
       import { useEffect, useState } from "react";
       import { useRouter } from "next/navigation";
       import { useAuthStore } from "@/lib/auth/store";

       export function AuthGuard({ children }: { children: React.ReactNode }) {
         const router = useRouter();
         const { isAuthenticated, tokenConfigured } = useAuthStore();
         const [hydrated, setHydrated] = useState(false);

         useEffect(() => {
           // zustand persist hydrates async from localStorage
           const unsub = useAuthStore.persist.onFinishHydration(() => {
             setHydrated(true);
           });
           // If already hydrated (e.g. not first render)
           if (useAuthStore.persist.hasHydrated()) {
             setHydrated(true);
           }
           return unsub;
         }, []);

         useEffect(() => {
           if (hydrated && isAuthenticated) {
             router.replace(tokenConfigured ? "/dashboard" : "/setup");
           }
         }, [hydrated, isAuthenticated, tokenConfigured, router]);

         // Show nothing until hydration completes (prevents flash)
         if (!hydrated) return null;
         // Authenticated users see nothing while redirect happens
         if (isAuthenticated) return null;
         return <>{children}</>;
       }
       ```

       Key detail: uses `useAuthStore.persist.onFinishHydration()` to avoid the flash-of-auth-page problem described in RESEARCH Pitfall 2. Shows nothing until hydration completes, then either redirects or shows children.
