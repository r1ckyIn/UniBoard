**1. Create `frontend/lib/hooks/use-count-up.ts`:**

Lightweight rAF-based number animation hook. Takes a target number (or null), returns the display number that animates from previous to new value over 400ms with ease-out cubic.

```typescript
"use client";
import { useState, useRef, useEffect } from "react";

export function useCountUp(target: number | null, duration = 400): number | null {
  const [display, setDisplay] = useState<number | null>(target);
  const prevRef = useRef<number | null>(target);

  useEffect(() => {
    if (target === null || prevRef.current === null) {
      setDisplay(target);
      prevRef.current = target;
      return;
    }
    const from = prevRef.current;
    const to = target;
    const start = performance.now();
    let raf: number;
    const animate = (now: number) => {
      const elapsed = now - start;
      const progress = Math.min(elapsed / duration, 1);
      const eased = 1 - Math.pow(1 - progress, 3); // ease-out cubic
      setDisplay(from + (to - from) * eased);
      if (progress < 1) raf = requestAnimationFrame(animate);
    };
    raf = requestAnimationFrame(animate);
    prevRef.current = target;
    return () => cancelAnimationFrame(raf);
  }, [target, duration]);

  return display;
}
```

```typescript
const { currentAvg, projectedFinal, assessedWeight } = useMemo(() => {
  let gradedSumSW = 0;
  let gradedSumW = 0;
  let totalSumSW = 0;
  let allFilled = true;

  assessments.forEach((a, i) => {
    if (a.score !== null) {
      gradedSumSW += a.score * a.weight;
      gradedSumW += a.weight;
      totalSumSW += a.score * a.weight;
    } else {
      const pred = predictions[i];
      if (pred !== null && pred !== undefined) {
        totalSumSW += (pred / a.max_score) * a.weight;
      } else {
        allFilled = false;
      }
    }
  });

  const totalWeight = assessments.reduce((s, a) => s + a.weight, 0);
  const currentAvg = gradedSumW > 0 ? (gradedSumSW / gradedSumW) * 100 : null;
  const projectedFinal = allFilled && totalWeight > 0 ? (totalSumSW / totalWeight) * 100 : null;

  return { currentAvg, projectedFinal, assessedWeight: gradedSumW };
}, [assessments, predictions]);
```
