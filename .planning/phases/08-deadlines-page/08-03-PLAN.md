```typescript
const deadlinesByDate = useMemo(() => {
  const map = new Map<string, Deadline[]>();
  for (const dl of deadlines) {
    const dateKey = format(new Date(dl.due_date), "yyyy-MM-dd");
    const existing = map.get(dateKey) ?? [];
    existing.push(dl);
    map.set(dateKey, existing);
  }
  return map;
}, [deadlines]);
```

```typescript
const filteredDeadlines = useMemo(() => {
  let list = allDeadlines;
  // Course filter
  if (selectedCourse) {
    list = list.filter(dl => dl.course_code === selectedCourse);
  }
  // Date filter (from calendar click)
  if (selectedDate) {
    list = list.filter(dl => format(new Date(dl.due_date), "yyyy-MM-dd") === selectedDate);
  }
  // Mode filter (All vs This Week)
  if (filterMode === "week") {
    list = list.filter(dl => {
      const days = differenceInCalendarDays(new Date(dl.due_date), new Date());
      return days >= 0 && days <= 7;
    });
  }
  // Sort by due_date ascending
  return [...list].sort((a, b) => new Date(a.due_date).getTime() - new Date(b.due_date).getTime());
}, [allDeadlines, selectedCourse, selectedDate, filterMode]);
```
