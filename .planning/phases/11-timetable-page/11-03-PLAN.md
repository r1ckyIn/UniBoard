       Current week computation:
       ```typescript
       const computeCurrentWeek = (weeks: SemesterWeek[]): number => {
         const now = new Date();
         const semesterStart = new Date("2026-02-23"); // Week 1 Monday
         const weeksDiff = differenceInWeeks(now, semesterStart);
         // Map to position 1-14, clamp
         const pos = Math.max(1, Math.min(14, weeksDiff + 1));
         // Verify this position exists in weeks array
         return weeks.find(w => w.position === pos) ? pos : 4; // fallback to 4
       };
       ```
