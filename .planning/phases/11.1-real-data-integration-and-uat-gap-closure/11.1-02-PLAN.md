**TimetablePage.tsx — Compute deadlineDays from all deadlines:**

Add a `deadlineDays` computation using the existing `allDeadlines` data. Add this useMemo after the existing deadline processing (around line 162):

```typescript
import { format as fnsFormat } from "date-fns";
// (format already imported as format, this is just confirming)

const deadlineDays = useMemo(() => {
  const dayMap = new Map<string, number>();
  for (const dl of allDeadlines) {
    if (dl.status === "graded") continue; // Skip completed deadlines
    const dateKey = format(new Date(dl.due_date), "yyyy-MM-dd");
    const current = dayMap.get(dateKey) ?? 0;
    dayMap.set(dateKey, current + (dl.weight ?? 0));
  }
  return Array.from(dayMap.entries()).map(([date, totalWeight]) => ({
    date,
    totalWeight,
  }));
}, [allDeadlines]);
```
