class _SessionCtx:

def _build_mock_factory(
    profiles: list[Profile],
    courses: list[Course],
    deadlines_by_tier: dict[str, list[UnifiedDeadline]],
) -> MagicMock:
    """Build a mock session factory for check_deadline_reminders.

    The function creates multiple sessions (one for profile query,
    one per user for courses+deadlines). We use side_effect to
    return different sessions for sequential calls.
    """
    # Session 1: returns profiles
    session1 = AsyncMock()
    profile_result = MagicMock()
    profile_result.scalars.return_value.all.return_value = profiles
    session1.execute = AsyncMock(return_value=profile_result)

    # Session 2+: per-user session for courses and deadline queries
    call_count = 0

    async def _execute_per_user(stmt: object) -> MagicMock:
        nonlocal call_count
        call_count += 1
        result = MagicMock()
        stmt_str = str(stmt)
        if "courses" in stmt_str.lower():
            result.scalars.return_value.all.return_value = courses
        elif "unified_deadlines" in stmt_str.lower():
            # Return all deadlines regardless of tier filtering
            # (the real DB filter handles tier windows; in test we
            #  return what would match)
            all_dls: list[UnifiedDeadline] = []
            for dls in deadlines_by_tier.values():
                all_dls.extend(dls)
            result.scalars.return_value.all.return_value = all_dls
        else:
            result.scalars.return_value.all.return_value = []
        return result

    session2 = AsyncMock()
    session2.execute = _execute_per_user
    session2.commit = AsyncMock()

    # NotificationService.create_notification mock
    notif_create_mock = AsyncMock(return_value=MagicMock())

    sessions = [session1, session2]
    idx = {"i": 0}

    def _factory() -> _SessionCtx:
        i = min(idx["i"], len(sessions) - 1)
        idx["i"] += 1
        return _SessionCtx(sessions[i])

    factory = MagicMock(side_effect=_factory)
    return factory, notif_create_mock
