Add method `agent_stream`:
```python
async def agent_stream(
    self,
    question: str,
    context_text: str,
    tools: list[dict],  # type: ignore[type-arg]
    tool_executor: Callable[[str, dict], Awaitable[str]],  # type: ignore[type-arg]
    language: str = "en",
    model: str = "claude-sonnet-4-20250514",
) -> AsyncGenerator[str, None]:
    """Agentic streaming loop with tool_use support.

    When Claude calls a tool (stop_reason="tool_use"), executes it via
    tool_executor, sends result back, and resumes streaming.
    """
    messages: list[dict] = [  # type: ignore[type-arg]
        {"role": "user", "content": f"Context:\n{context_text}\n\nQuestion: {question}"}
    ]

    max_iterations = 5  # Safety limit on tool call loops
    for _ in range(max_iterations):
        async with self._client.messages.stream(
            model=model,
            max_tokens=1000,
            system=get_qa_prompt(language),
            messages=messages,
            tools=tools,
        ) as stream:
            async for event in stream:
                if event.type == "text":
                    yield event.text

            final_message = await stream.get_final_message()

            if final_message.stop_reason == "tool_use":
                # Append assistant message with tool_use blocks
                messages.append({"role": "assistant", "content": final_message.content})

                # Execute each tool call
                tool_results = []
                for block in final_message.content:
                    if block.type == "tool_use":
                        result = await tool_executor(block.name, block.input)
                        tool_results.append({
                            "type": "tool_result",
                            "tool_use_id": block.id,
                            "content": result,
                        })

                messages.append({"role": "user", "content": tool_results})
                continue
            else:
                # stop_reason is "end_turn" -- done
                break
```

                course_id=course_id,
                language=lang,
            ):
                yield {"event": "token", "data": json.dumps({"text": token})}

            yield {"event": "done", "data": json.dumps({"status": "complete"})}
        except Exception as exc:
            yield {"event": "error", "data": json.dumps({"message": str(exc)})}

    return EventSourceResponse(event_generator(), ping=15)
