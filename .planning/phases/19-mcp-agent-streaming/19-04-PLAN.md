**1. Create `frontend/components/settings/LanguageSection.tsx`:**

Language preference dropdown with English/中文 options. On change: (1) persist to backend via PATCH /users/me, (2) auto-switch next-intl locale.

```typescript
"use client";

import { useState, useCallback } from "react";
import { useTranslations, useLocale } from "next-intl";
import { useRouter, usePathname } from "@/lib/i18n/routing";
import { useCurrentUser, useUpdateUser } from "@/hooks/use-user";
import { Globe } from "lucide-react";

const LANGUAGE_OPTIONS = [
  { value: "en", label: "English" },
  { value: "zh", label: "中文" },
] as const;

export default function LanguageSection() {
  const t = useTranslations("settings");
  const locale = useLocale();
  const router = useRouter();
  const pathname = usePathname();
  const { data: user } = useCurrentUser();
  const updateUser = useUpdateUser();

  // Use profile language_preference if available, else fall back to current locale
  const currentLang = user?.data?.language_preference || locale;
  const [selectedLang, setSelectedLang] = useState(currentLang);

  const handleLanguageChange = useCallback(
    async (lang: string) => {
      setSelectedLang(lang);

      // 1. Persist to backend
      updateUser.mutate(
        { language_preference: lang } as Record<string, unknown>,
        {
          onSuccess: () => {
            // 2. Switch frontend locale via next-intl routing
            router.replace(pathname, { locale: lang });
          },
        },
      );
    },
    [updateUser, router, pathname],
  );

  return (
    <div>
      <div className="flex items-center gap-[8px] mb-[12px]">
        <Globe size={18} className="text-[#d97757]" />
        <span className="text-[0.82rem] text-[#6b6b65]">
          {t("language.description")}
        </span>
      </div>

      <div className="flex gap-[10px]">
        {LANGUAGE_OPTIONS.map((option) => (
          <button
            key={option.value}
            type="button"
            onClick={() => handleLanguageChange(option.value)}
            className={`px-[20px] py-[10px] rounded-[10px] text-[0.84rem] font-semibold border-[1.5px] transition-all duration-200 ${
              selectedLang === option.value
                ? "border-[#d97757] bg-[rgba(217,119,87,0.08)] text-[#d97757]"
                : "border-[#e8e5dd] bg-[#f6f5f0] text-[#6b6b65] hover:border-[#d0cdc4]"
            }`}
          >
            {option.label}
          </button>
        ))}
      </div>

      {updateUser.isPending && (
        <p className="text-[0.72rem] text-[#9b9b94] mt-[8px] italic">
          {t("language.saving")}
        </p>
      )}
    </div>
  );
}
```

Note: The `useUpdateUser` mutation from `use-user.ts` uses PATCH /users/me which now accepts `language_preference` (added in Plan 01 schemas). If `useUpdateUser` types are too strict (only allowing known fields), the executor should check `UpdateUserBody` type from the OpenAPI types and update accordingly. The `as Record<string, unknown>` cast may be needed if the generated type doesn't include `language_preference` yet.
