```typescript
type Platform = "canvas" | "ed";
const STORAGE_KEY = "uniboard.setup.tokenValidated";
const TTL_MS = 5 * 60 * 1000;

export async function hashToken(token: string): Promise<string> {
  const enc = new TextEncoder().encode(token);
  const buf = await crypto.subtle.digest("SHA-256", enc);
  return Array.from(new Uint8Array(buf))
    .map((b) => b.toString(16).padStart(2, "0"))
    .join("");
}

interface CacheEntry { hash: string; validatedAt: string; }
interface CacheShape { [platform: string]: CacheEntry; }

export function readTokenCache(): CacheShape;       // SSR-safe (returns {} if no window)
export function writeTokenCache(platform: Platform, hash: string): void;
export function clearTokenCache(platform?: Platform): void;
export async function isTokenCachedAndFresh(platform: Platform, token: string): Promise<boolean>;
```

async function submitWithBackoff(submit: () => Promise<void>): Promise<void> {
  const delays = [2000, 4000, 8000];
  for (let i = 0; i <= delays.length; i++) {
    try {
      await submit();
      return;
    } catch (err) {
      const status = (err as { status?: number })?.status;
      const isRetryable = !status || status >= 500;
      if (!isRetryable || i === delays.length) throw err;
      await new Promise((r) => setTimeout(r, delays[i]));
    }
  }
}
