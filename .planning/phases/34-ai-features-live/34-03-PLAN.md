<math_reference>
<!-- From RESEARCH §7 — full closed-form derivation -->

Given:
- `current_wam`: existing weighted average from completed courses (Decimal)
- `cp_done`: total credit points completed (Decimal)
- `cp_remain`: user-input remaining credit points (Decimal)
- `target_wam`: user-input target WAM (Decimal)

The key identity:
```
target_wam = (current_wam * cp_done + required_avg * cp_remain) / (cp_done + cp_remain)
```

Solve for `required_avg`:
```
required_avg = (target_wam * (cp_done + cp_remain) - current_wam * cp_done) / cp_remain
```

`max_reachable` (assume score 100 on remaining):
```
max_reachable = (current_wam * cp_done + 100 * cp_remain) / (cp_done + cp_remain)
```

Edge cases:
- `cp_remain == 0` → required_avg undefined; report current_wam as max_reachable; is_achievable = (current_wam >= target)
- `current_wam >= target` → required_avg = 0 (any future score keeps target); max_reachable = computed
- `required_avg > 100` → is_achievable = False; compute suggested_target = highest USYD band <= original_target where band <= max_reachable

USYD bands: HD=85, Distinction=75, Credit=65, Pass=50

    async def calculate_multi_course_path(
        self,
        user_id: uuid.UUID,
        target_wam: float,
        remaining_credit_points: int,
    ) -> MultiCoursePathResponse:
        """Compute required average for remaining REMAINING units (not assessments).

        Math (closed-form, per RESEARCH §7):
            target_wam = (current_wam * cp_done + required_avg * cp_remain)
                          / (cp_done + cp_remain)
            => required_avg = (target_wam * (cp_done + cp_remain)
                               - current_wam * cp_done) / cp_remain

        Returns advisory_text=None — call get_path_advisory separately for AI line.
        Per phase 34 AIFEAT-03 / D-C3 / D-D1.
        """
        summary = await self.get_summary(user_id)
        current_wam = Decimal(str(summary.cumulative_wam))
        cp_done = Decimal(str(summary.total_credit_points))
        cp_remain = Decimal(str(remaining_credit_points))
        target = Decimal(str(target_wam))

        # Edge case: 0 remaining credits
        if cp_remain == 0:
            return MultiCoursePathResponse(
                target_wam=float(target),
                current_wam=float(current_wam),
                is_achievable=current_wam >= target,
                required_avg=None,
                max_reachable=float(current_wam),
                suggested_target=None,
                advisory_text=None,  # caller invokes get_path_advisory separately
                language=self._language,
            )

        # Edge case: target already met
        if current_wam >= target:
            max_reachable = (
                (current_wam * cp_done + Decimal("100") * cp_remain)
                / (cp_done + cp_remain)
            ).quantize(_TWO_PLACES, rounding=ROUND_HALF_UP)
            return MultiCoursePathResponse(
                target_wam=float(target),
                current_wam=float(current_wam),
                is_achievable=True,
                required_avg=0.0,  # any future score keeps target
                max_reachable=float(max_reachable),
                suggested_target=None,
                advisory_text=None,
                language=self._language,
            )

        total_cp = cp_done + cp_remain
        required = (
            (target * total_cp - current_wam * cp_done) / cp_remain
        ).quantize(_TWO_PLACES, rounding=ROUND_HALF_UP)

        max_reachable = (
            (current_wam * cp_done + Decimal("100") * cp_remain) / total_cp
        ).quantize(_TWO_PLACES, rounding=ROUND_HALF_UP)

        if required > Decimal("100"):
            # Unreachable — find next-best band
            suggested = self._suggest_next_band(max_reachable, target)
            return MultiCoursePathResponse(
                target_wam=float(target),
                current_wam=float(current_wam),
                is_achievable=False,
                required_avg=None,
                max_reachable=float(max_reachable),
                suggested_target=float(suggested) if suggested is not None else None,
                advisory_text=None,
                language=self._language,
            )

        return MultiCoursePathResponse(
            target_wam=float(target),
            current_wam=float(current_wam),
            is_achievable=True,
            required_avg=float(required),
            max_reachable=float(max_reachable),
            suggested_target=None,
            advisory_text=None,
            language=self._language,
        )

    @staticmethod
    def _suggest_next_band(
        max_reachable: Decimal,
        original_target: Decimal,
    ) -> Decimal | None:
        """Pick the highest USYD band still achievable AND below the original target.

        Per phase 34 D-C3: avoid dead-end UX; suggest the next reachable goal.
        """
        for band in _USYD_BANDS:
            if band < original_target and max_reachable >= band:
                return band
        return None

    async def get_path_advisory(
        self,
        result: MultiCoursePathResponse,
        course_levels: list[int] | None = None,
    ) -> str | None:
        """Wrap math result in 30-50 word AI verdict + tactic.

        Returns None on AI failure (D-D1 silent fallback). Sentry-tagged
        with phase=34 + feature=path_planner.
        """
        if not self._anthropic_api_key:
            return None
        try:
            from src.services.ai_engine import AIEngine

            ai = AIEngine(api_key=self._anthropic_api_key)
            system_prompt = get_path_advisory_prompt(result.language)
            user_msg = get_path_advisory_user_message(
                is_achievable=result.is_achievable,
                required_avg=result.required_avg,
                max_reachable=result.max_reachable,
                suggested_target=result.suggested_target,
                course_levels=course_levels,
                language=result.language,
            )
            response = await ai.ask_question(
                question=user_msg,
                context_text="",
                system_prompt=system_prompt,
            )
            return response.answer.strip()
        except Exception:
            with sentry_sdk.new_scope() as scope:
                scope.set_tag("phase", "34")
                scope.set_tag("feature", "path_planner")
                sentry_sdk.capture_exception()
            logger.warning(
                "path_advisory_ai_failed",
                target_wam=result.target_wam,
                exc_info=True,
            )
            return None
    ```

    # Phase 34 AIFEAT-03 — multi-course path planner with optional AI advisory
    @router.post("/multi-course-path")
    @limiter.limit("10/minute")  # involves AI call; matches /gpa/path rate
    async def calculate_multi_course_path(
        body: MultiCoursePathRequest,
        request: Request,
        session: AsyncSession = Depends(get_session),
        current_user_id: uuid.UUID = Depends(get_current_user_id),
        svc: GPAService = Depends(get_gpa_service),
    ) -> SuccessResponse[MultiCoursePathResponse]:
        """Calculate required avg for remaining UNITS to reach target WAM.

        Per phase 34 AIFEAT-03:
        - Math always returned (Decimal-precise via service)
        - AI advisory_text=None on failure (D-D1 silent fallback)
        - Pydantic validates target_wam in [0, 100], remaining_credit_points >= 0

        Returns SuccessResponse[MultiCoursePathResponse].
        """
        # 1. Compute math (always succeeds)
        result = await svc.calculate_multi_course_path(
            current_user_id,
            target_wam=body.target_wam,
            remaining_credit_points=body.remaining_credit_points,
        )

        # 2. Optional AI advisory (D-D1 fallback — None on failure)
        advisory_text = await svc.get_path_advisory(result)

        # 3. Wire advisory back into response (math-only result was advisory_text=None)
        result_with_advisory = result.model_copy(update={"advisory_text": advisory_text})

        meta = get_request_meta(request)
        return SuccessResponse(data=result_with_advisory, meta=meta)
    ```
