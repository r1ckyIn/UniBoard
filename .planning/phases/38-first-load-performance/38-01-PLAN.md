    ```tsx
              const coursesCache = queryClient.getQueryData<{ data: Array<{ id: string; code: string }> }>(
                courseOptions.list().queryKey,
              );
              nearestCourseId = coursesCache?.data.find((c) => c.code === nearestCode)?.id ?? null;
            }
          } catch (e) {
            wrapSentry("deadlines-upcoming", userId)(e);
          }

          // Wave C: if nearest course identified, prefetch its detail in parallel with remainder
          const finalPrefetches: Promise<unknown>[] = [...independentPrefetches];
          if (nearestCourseId) {
            finalPrefetches.push(
              queryClient.prefetchQuery({
                ...courseOptions.detail(nearestCourseId),
                queryFn: () => api.get(`courses/${nearestCourseId}`).json(),
              }).catch(wrapSentry("course-detail", userId, { courseId: nearestCourseId })),
            );
          }
          await Promise.allSettled(finalPrefetches);
        },
      });
    }
    ```
