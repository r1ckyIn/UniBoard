    ```tsx
    import { Suspense } from "react";
    import { setRequestLocale } from "next-intl/server";
    import PredictPage from "@/components/predict/PredictPage";
    import { createPrefetchedPage, wrapSentry } from "@/lib/rsc/create-prefetched-page";
    import { getServerApiClient } from "@/lib/rsc/server-query-fn";
    import { courseOptions } from "@/hooks/use-courses";
    import { roiOptions } from "@/hooks/use-roi";

    export const dynamic = "force-dynamic";

    type Props = { params: Promise<{ locale: string }> };

    export default async function PredictRoute({ params }: Props) {
      const { locale } = await params;
      setRequestLocale(locale);

      return await createPrefetchedPage({
        children: (
          <Suspense>
            <PredictPage />
          </Suspense>
        ),
        run: async ({ queryClient, accessToken, userId }) => {
          const api = await getServerApiClient(accessToken);

          // Step 1: await /courses (N source of truth)
          let courseIds: string[] = [];
          try {
            const courses = await queryClient.fetchQuery({
              ...courseOptions.list(),
              queryFn: () => api.get("courses").json(),
            });
            courseIds = (courses as { data?: Array<{ id: string }> })
              .data?.map((c) => c.id) ?? [];
          } catch (e) {
            wrapSentry("courses", userId)(e);
          }

          // Step 2: full N-fanout — ALL N × detail + ALL N × roi in parallel (D-B3)
          if (courseIds.length > 0) {
            const prefetches: Promise<unknown>[] = [
              ...courseIds.map((id) =>
                queryClient
                  .prefetchQuery({
                    ...courseOptions.detail(id),
                    queryFn: () => api.get(`courses/${id}`).json(),
                  })
                  .catch(wrapSentry("course-detail", userId, { courseId: id })),
              ),
              ...courseIds.map((id) =>
                queryClient
                  .prefetchQuery({
                    ...roiOptions.course(id),
                    queryFn: () => api.get(`courses/${id}/roi`).json(),
                  })
                  .catch(wrapSentry("roi", userId, { courseId: id })),
              ),
            ];
            await Promise.allSettled(prefetches);
          }
        },
      });
    }
    ```

    Rationale: D-B3 locks full N-fanout — ALL N × detail AND ALL N × roi in ONE `Promise.allSettled`. Do NOT cap, do NOT batch. Accepted tradeoff: cold-start amplification (P03 characterises; warmup mitigates if p95 > 2s).

    Preserve the existing `<Suspense>` wrapper (inner) — HOF wraps HydrationBoundary outer.

    }
