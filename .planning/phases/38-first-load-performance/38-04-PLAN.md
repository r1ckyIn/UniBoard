    Step 1 — Create `frontend/tests/e2e/perf/helpers/auth.ts`:
    ```typescript
    import type { Page } from "@playwright/test";

    const EMAIL = "perf-test@uniboard.uk";

    /**
     * Log in the perf-test user by posting to Supabase Auth directly, then
     * inject the resulting session cookies into the Playwright browser context.
     *
     * The PERF_TEST_PASSWORD env var MUST be set (GitHub Secret in CI, local
     * .env.local for dev). Fails fast if missing so we never accidentally run
     * a blank-password login attempt.
     */
    export async function loginAsPerfTestUser(page: Page): Promise<void> {
      const password = process.env.PERF_TEST_PASSWORD;
      if (!password) {
        throw new Error(
          "PERF_TEST_PASSWORD not set — add to GitHub Secrets (CI) or frontend/.env.local (dev).",
        );
      }

      const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL;
      const supabaseAnonKey =
        process.env.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY ??
        process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY;
      if (!supabaseUrl || !supabaseAnonKey) {
        throw new Error("Supabase URL / anon key not set in env");
      }

      const resp = await page.request.post(
        `${supabaseUrl}/auth/v1/token?grant_type=password`,
        {
          headers: {
            apikey: supabaseAnonKey,
            "Content-Type": "application/json",
          },
          data: { email: EMAIL, password },
        },
      );
      if (!resp.ok()) {
        const body = await resp.text();
        throw new Error(`Login failed ${resp.status()}: ${body}`);
      }
      const { access_token, refresh_token } = await resp.json();

      // Use the app's own /api/v1 cookie setter route if available; otherwise
      // directly set the Supabase session cookies. Phase 33 established
      // client-side persistence; replicate that cookie shape here.
      await page.context().addCookies([
        {
          name: "sb-access-token",
          value: access_token,
          url: page.url() || "http://localhost:3001",
        },
        {
          name: "sb-refresh-token",
          value: refresh_token,
          url: page.url() || "http://localhost:3001",
        },
      ]);

      // Some @supabase/ssr deployments serialise session via a different cookie
      // key. If this simple setter does not persist session across navigation,
      // fall back to visiting /auth?mode=login and filling the form — the
      // helper is allowed to do that fallback; just keep it in this file.
    }
    ```

    IMPORTANT — cookie shape verification: `@supabase/ssr` uses base64 chunked cookies (e.g., `sb-<project-ref>-auth-token`). The naive `sb-access-token` approach above may NOT work against the real SSR reader. If the first test run shows the middleware still redirects to `/auth`, SWITCH to the "visit /auth, fill form, click submit" UI login flow. Both approaches are acceptable — pick what works.
